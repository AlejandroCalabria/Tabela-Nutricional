spring.application.name=nutriiftm

# Profile ativo por padrao = dev (em producao, sobrescrever com env var SPRING_PROFILES_ACTIVE=prod)
spring.profiles.active=${SPRING_PROFILES_ACTIVE:dev}

# Datasource: valores assumidos via variavel de ambiente, com fallback para dev local
spring.datasource.url=jdbc:mysql://${DB_HOST:localhost}:${DB_PORT:3306}/${DB_NAME:nutricional}
spring.datasource.username=${DB_USER:root}
spring.datasource.password=${DB_PASSWORD:}
spring.datasource.driver-class-name=com.mysql.cj.jdbc.Driver

spring.jpa.open-in-view=false
spring.jpa.properties.hibernate.dialect=org.hibernate.dialect.MySQLDialect
spring.jpa.hibernate.naming.physical-strategy=org.hibernate.boot.model.naming.PhysicalNamingStrategyStandardImpl

# Flyway controla o schema; Hibernate so valida que o schema bate com as entidades
spring.jpa.hibernate.ddl-auto=validate
spring.flyway.enabled=true
spring.flyway.baseline-on-migrate=true
spring.flyway.locations=classpath:db/migration
